total = len(sentences)
ok, failed = 0, []
print(f"開始錄製 {total} 句…\n")

for i, (fname, text) in enumerate(sentences.items(), 1):
    path = os.path.join(OUT_DIR, fname)
    done = False
    for attempt in range(1, 4):                      # 失敗重試 3 次，等待遞增
        try:
            gTTS(text=text, lang="es", tld="com.mx").save(path)
            size = os.path.getsize(path)
            if size < 500:
                raise ValueError(f"檔案只有 {size} bytes，可能是空檔")
            print(f"  [{i:>3}/{total}] ✅ {fname:<44} {text}")
            ok += 1
            done = True
            break
        except Exception as e:
            wait = attempt * 3
            print(f"  [{i:>3}/{total}] ⚠️ 第{attempt}次失敗（{e}），{wait}秒後重試…")
            time.sleep(wait)
    if not done:
        print(f"  [{i:>3}/{total}] ❌ 放棄：{fname}")
        failed.append(fname)
    time.sleep(random.uniform(0.5, 1.2))             # 避免被判定成批次攻擊

print(f"\n完成：成功 {ok} / 失敗 {len(failed)}")
if failed:
    print("失敗清單（這幾個要再跑一次）：")
    for f in failed:
        print("   " + f)

with zipfile.ZipFile(ZIP_NAME, "w", zipfile.ZIP_DEFLATED) as z:
    for f in sorted(os.listdir(OUT_DIR)):
        z.write(os.path.join(OUT_DIR, f), f)
