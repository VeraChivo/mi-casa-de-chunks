out_dir = "chunks_batch2_part3"
os.makedirs(out_dir, exist_ok=True)
failed = []
total = len(sentences)
for i, (fname, text) in enumerate(sentences.items(), 1):
    path = os.path.join(out_dir, fname)
    print(f"[{i}/{total}] 錄製中：{fname} ← {text}")
    ok = False
    for attempt in range(1, 4):
        try:
            gTTS(text=text, lang='es', tld='com.mx').save(path)
            if os.path.getsize(path) > 500:
                ok = True
                break
            print(f"   ⚠️ 檔案太小，重試 {attempt}/3")
        except Exception as e:
            print(f"   ⚠️ 失敗（{e}），重試 {attempt}/3")
        time.sleep(2 * attempt)
    print("   ✅ 完成" if ok else "   ❌ 放棄")
    if not ok:
        failed.append(fname)
    time.sleep(random.uniform(0.5, 1.2))

zip_path = f"audio_chunks_batch2_part3.zip"
with zipfile.ZipFile(zip_path, 'w') as z:
    for f in os.listdir(out_dir):
        fp = os.path.join(out_dir, f)
        if os.path.getsize(fp) > 500:
            z.write(fp, f)

print("\n==============================")
print(f"成功 {total - len(failed)} / {total}")
if failed:
    print("❌ 失敗清單（之後再補錄）：", failed)
print("zip 路徑：", os.path.abspath(zip_path))
print("zip 大小：", os.path.getsize(zip_path), "bytes")
print("如果下載視窗沒跳出來：左側📁檔案面板 → 對 zip 按右鍵 → 下載")

from google.colab import files
files.download(zip_path)
