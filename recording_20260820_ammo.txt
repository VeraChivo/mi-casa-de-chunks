os.makedirs(OUT, exist_ok=True)
for sub in ['ammo']:
    os.makedirs(os.path.join(OUT, sub), exist_ok=True)

total = len(sentences)
done, failed = 0, []

for i, (path, text) in enumerate(sentences.items(), 1):
    target = os.path.join(OUT, path)
    print(f"[{i}/{total}] {path}  ←  {text}")
    for attempt in range(1, 4):
        try:
            gTTS(text=text, lang="es", tld="com.mx").save(target)
            size = os.path.getsize(target)
            if size < 800:
                raise RuntimeError(f"檔案只有 {size} bytes，可能是空檔")
            done += 1
            break
        except Exception as e:
            print(f"    ⚠️ 第 {attempt} 次失敗：{e}")
            if attempt == 3:
                failed.append(path)
            else:
                time.sleep(attempt * 3)   # 3秒 → 6秒，遞增等待
    # 每句之間隨機停頓，避免被 Google 判定成批次攻擊整批擋掉
    time.sleep(random.uniform(0.5, 1.2))

print("\n" + "=" * 50)
print(f"完成 {done}/{total} 句")
if failed:
    print(f"❌ 失敗 {len(failed)} 句（可以只把這幾句挑出來重跑）：")
    for f in failed:
        print("   " + f)

zip_path = "/content/audio_ammo_20260820.zip"
with zipfile.ZipFile(zip_path, "w", zipfile.ZIP_DEFLATED) as z:
    for root, _, files in os.walk(OUT):
        for f in files:
            full = os.path.join(root, f)
            z.write(full, os.path.relpath(full, OUT))

print(f"\n📦 打包完成：{os.path.abspath(zip_path)}")
