import os, time, random, zipfile
from gtts import gTTS

OUT_DIR = "corazon"
ZIP_NAME = "corazon_audio_20260911.zip"
os.makedirs(OUT_DIR, exist_ok=True)

# 檔名 → 要唸的西語
sentences = {
    "corazon_crecimiento_06.mp3": "Mi única ambición es ser libre toda mi vida.",
}

total = len(sentences)
ok, failed = 0, []
print(f"開始錄製 {total} 句…\n")

for i, (fname, text) in enumerate(sentences.items(), 1):
    path = os.path.join(OUT_DIR, fname)
    done = False
    for attempt in range(1, 4):                      # 失敗重試 3 次，等待遞增
        try:
            gTTS(text=text, lang="es", tld="com.mx").save(path)
            size = os.path.getsize(path)
            if size < 500:
                raise ValueError(f"檔案只有 {size} bytes，可能是空檔")
            print(f"  [{i:>3}/{total}] ✅ {fname:<44} {text}")
            ok += 1
            done = True
            break
        except Exception as e:
            wait = attempt * 3
            print(f"  [{i:>3}/{total}] ⚠️ 第{attempt}次失敗（{e}），{wait}秒後重試…")
            time.sleep(wait)
    if not done:
        print(f"  [{i:>3}/{total}] ❌ 放棄：{fname}")
        failed.append(fname)
    time.sleep(random.uniform(0.5, 1.2))             # 避免被判定成批次攻擊

print(f"\n完成：成功 {ok} / 失敗 {len(failed)}")
if failed:
    print("失敗清單（這幾個要再跑一次）：")
    for f in failed:
        print("   " + f)

with zipfile.ZipFile(ZIP_NAME, "w", zipfile.ZIP_DEFLATED) as z:
    for f in sorted(os.listdir(OUT_DIR)):
        z.write(os.path.join(OUT_DIR, f), f)

zip_path = os.path.abspath(ZIP_NAME)
print(f"\n📦 打包完成：{zip_path}")
print(f"   檔案大小：{os.path.getsize(ZIP_NAME)/1024/1024:.2f} MB")
print("   若下面的自動下載沒跳出來，就到 Colab 左側檔案面板照上面路徑手動右鍵下載。")

from google.colab import files
files.download(ZIP_NAME)
